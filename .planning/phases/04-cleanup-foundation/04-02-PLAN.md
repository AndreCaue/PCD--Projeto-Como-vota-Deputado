```typescript
import { useEffect, useState } from "react";
import { patrimonioService, type BemPatrimonio } from "@/services/api";

export function useBensPatrimonio(deputiesId: string, options?: { ano?: number; tipo?: string }) {
  const [data, setData] = useState<BemPatrimonio[] | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
  useEffect(() => {
    if (!deputiesId) return;
    const fetchData = async () => {
      setLoading(true); setError(null);
      try {
        const result = await patrimonioService.buscar(deputiesId, options?.ano, options?.tipo);
        setData(result);
      } catch (e: any) { setError(e.message || "Erro ao carregar patrimônio"); }
      finally { setLoading(false); }
    };
    fetchData();
  }, [deputiesId, options?.ano, options?.tipo]);
  return { data, loading, error };
}
```
